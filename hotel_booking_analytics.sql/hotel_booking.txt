1. Occupancy rate by month and city

 select count(b.booking_id) as total_bookings from h.city, date_format(b.checkin_date, '%y-%m') as month, bookings b join hotels h on b.hotel_id = h.hotel_id group by h.city, date_format(b.checkin_date, '%y-%m') order by month, h.city;

3. Revenue per available room (RevPAR)

select h.name as hotel_name, h.city, sum(b.total_amount) as total_revenue, count(b.booking_id) as total_bookings, round(sum(b.total_amount) / count(b.booking_id), 2) as revenue_per_room from bookings b join hotels h on b.hotel_id = h.hotel_id group by h.hotel_id, h.name, h.city order by revenue_per_room desc;
